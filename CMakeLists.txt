cmake_minimum_required(VERSION 3.21)
project(dlssg_turing VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This component requires Windows x64 and MSVC (SEH and Windows SDK).")
endif()
include(CTest)
set(TRP_NGX_SDK_DIR "" CACHE PATH "Locally supplied NGX SDK root containing include/nvsdk_ngx.h")
if(NOT EXISTS "${TRP_NGX_SDK_DIR}/include/nvsdk_ngx.h")
    message(FATAL_ERROR "Set TRP_NGX_SDK_DIR to your NGX SDK; no vendor files are downloaded or bundled.")
endif()
add_library(dlssg_turing STATIC extern/MFGAmpere/runtime.cpp
    extern/RTX40MFG/midpoint_fix.cpp extern/RTX40MFG/dlssg_provider_policy.cpp)
add_library(dlssg_turing::compat ALIAS dlssg_turing)
target_compile_features(dlssg_turing PUBLIC cxx_std_23)
target_compile_definitions(dlssg_turing PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(dlssg_turing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${TRP_NGX_SDK_DIR}/include")
target_link_libraries(dlssg_turing PUBLIC d3d12 dxgi bcrypt version)
if(BUILD_TESTING)
    add_subdirectory(tests/ampere-runtime)
    add_subdirectory(tests/turing)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME PinnedSource COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_source.py")
    endif()
endif()
